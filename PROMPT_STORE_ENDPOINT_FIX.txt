IMPLEMENTATION:
```typescript
import { NextRequest } from "next/server";
import { ok, err } from "@/lib/response";
import prisma from "@/lib/prisma";

interface Params {
  params: Promise<{ slug: string }>;
}

export async function GET(_req: NextRequest, { params }: Params) {
  try {
    const { slug } = await params;

    const store = await prisma.store.findUnique({
      where: {
        slug,
        isActive: true,
      },
      select: {
        id: true,
        name: true,
        slug: true,
        description: true,
        logoUrl: true,
        bannerUrl: true,
        province: true,
        city: true,
        district: true,
        address: true,
        phone: true,
        rating: true,
        totalSold: true,
        isVerified: true,
        isActive: true,
        createdAt: true,
        _count: {
          select: {
            products: true,
          },
        },
      },
    });

    if (!store) {
      return err("Toko tidak ditemukan", 404);
    }

    return ok(store);
  } catch (error) {
    console.error("[API] Error getting store:", error);
    return err("Internal server error", 500);
  }
}
```
